cmake_minimum_required(VERSION 3.28)

#
# 1. 准备阶段
#

# 使用vcpkg进行包管理
if(ANDROID)
    if (ANDROID_ABI MATCHES "arm64-v8a")
        set(VCPKG_TARGET_TRIPLET "arm64-android" CACHE STRING "" FORCE)
    elseif(ANDROID_ABI MATCHES "armeabi-v7a")
        set(VCPKG_TARGET_TRIPLET "arm-android" CACHE STRING "" FORCE)
    elseif(ANDROID_ABI MATCHES "x86_64")
        set(VCPKG_TARGET_TRIPLET "x64-android" CACHE STRING "" FORCE)
    else()
        message(FATAL_ERROR "安卓当前仅支持arm64-v8a、armeabi-v7a和x86_64架构")
    endif()
	if(ONLYCONSOLE)
		message(FATAL_ERROR "安卓不支持仅编译tjs2")
	endif()
    set(CMAKE_ANDROID_NDK $ENV{ANDROID_NDK_HOME})
    set(VCPKG_CHAINLOAD_TOOLCHAIN_FILE $ENV{ANDROID_NDK_HOME}/build/cmake/android.toolchain.cmake)
elseif(LINUX)
    set(VCPKG_TARGET_TRIPLET "x64-linux-release")
elseif(EMSCRIPTEN)
    set(VCPKG_TARGET_TRIPLET "wasm32-emscripten")
elseif(WINDOWS)
    if(MSVC)
        add_compile_options("$<$<CXX_COMPILER_ID:MSVC>:/utf-8>")
        if(CMAKE_GENERATOR_PLATFORM STREQUAL "x64")
            set(VCPKG_TARGET_TRIPLET "x64-windows")
        elseif(CMAKE_GENERATOR_PLATFORM STREQUAL "Win32")
            set(VCPKG_TARGET_TRIPLET "x86-windows")
        endif()
    else()
        set(VCPKG_TARGET_TRIPLET "x64-mingw-dynamic")
    endif()
elseif(MACOS)
    if(NOT DEFINED VCPKG_TARGET_TRIPLET)
        if(CMAKE_OSX_ARCHITECTURES MATCHES "x86_64" AND NOT CMAKE_OSX_ARCHITECTURES MATCHES "arm64")
            set(VCPKG_TARGET_TRIPLET "x64-osx" CACHE STRING "" FORCE)
        else()
            set(VCPKG_TARGET_TRIPLET "arm64-osx" CACHE STRING "" FORCE)
        endif()
    endif()
elseif(IOS)
    if(ONLYCONSOLE)
        message(FATAL_ERROR "iOS 不支持仅编译 tjs2")
    endif()
    set(CMAKE_SYSTEM_NAME iOS CACHE STRING "" FORCE)
    if(NOT DEFINED VCPKG_TARGET_TRIPLET)
        if(CMAKE_OSX_SYSROOT MATCHES "iphonesimulator")
            if(CMAKE_OSX_ARCHITECTURES MATCHES "x86_64")
                set(VCPKG_TARGET_TRIPLET "x64-ios-simulator" CACHE STRING "" FORCE)
            else()
                set(VCPKG_TARGET_TRIPLET "arm64-ios-simulator" CACHE STRING "" FORCE)
            endif()
        else()
            set(VCPKG_TARGET_TRIPLET "arm64-ios" CACHE STRING "" FORCE)
        endif()
    endif()
endif()
if(OHOS)
    # OHOS: 预先编译第三方依赖库
    set(OHOS_DEPS_SCRIPT "${CMAKE_SOURCE_DIR}/ohos/build-ohos.ps1")
    set(OHOS_DEPS_STAMP "${CMAKE_SOURCE_DIR}/out/ohos/deps/install/${OHOS_ARCH}/.stamp")
    if(NOT EXISTS "${OHOS_DEPS_STAMP}")
        message(STATUS "OHOS: Building dependencies (one-time)...")
        execute_process(
            COMMAND powershell -ExecutionPolicy Bypass -File "${OHOS_DEPS_SCRIPT}" deps -Arch ${OHOS_ARCH}
            WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}"
            RESULT_VARIABLE _ohos_deps_rc
        )
        if(NOT _ohos_deps_rc EQUAL 0)
            message(WARNING "OHOS deps build returned ${_ohos_deps_rc}")
        endif()
    endif()
else()
    set(CMAKE_TOOLCHAIN_FILE $ENV{VCPKG_ROOT}/scripts/buildsystems/vcpkg.cmake)
endif()

# SDL版本选择: ON=SDL3, OFF=SDL2
option(USE_SDL3 "Use SDL3 (OFF for SDL2)" ON)
if((ANDROID OR IOS) AND NOT USE_SDL3)
    # android/iOS由于存在平台入口与触屏层，所以不支持SDL2
    message(FATAL_ERROR "安卓/iOS 不支持 SDL2 作为硬件后端")
endif()
# 视频解码后端: ON=FFmpeg, OFF=系统API(MediaFoundation/MediaCodec)
# 感觉没必要禁用，OFF的实现基本没用༼ ༎ຶ ෴ ༎ຶ༽
option(USE_FFMPEG "Use FFmpeg for video/audio decoding" ON)
# 图形后端编译选项
option(USE_RENDER_GL "OpenGL 3.3+/OpenGLES 3.0+ backend" ON)
option(USE_RENDER_VK "Vulkan backend" ON)
set(KRKR_METAL_DEFAULT OFF)
if((IOS OR MACOS) AND USE_SDL3 AND NOT ONLYCONSOLE)
    set(KRKR_METAL_DEFAULT ON)
endif()
option(USE_RENDER_METAL "Native Metal backend (Apple SDL3 only)" ${KRKR_METAL_DEFAULT})
if(USE_RENDER_METAL AND (NOT (IOS OR MACOS) OR NOT USE_SDL3 OR ONLYCONSOLE))
    message(FATAL_ERROR "USE_RENDER_METAL requires an Apple SDL3 full runtime")
endif()
option(KRKR_HOST_LIBRARY "Build KRKR as an embeddable iOS framework" OFF)
# 宿主共享 SDL3：指向某个 SDL3.framework（单一架构切片）时，KRKRRuntime 动态链接它，
# 不再静态并入 vcpkg 的 SDL3。宿主 App 负责嵌入同一个 SDL3.framework，供其它引擎复用。
set(KRKR_SDL3_FRAMEWORK "" CACHE PATH "iOS host: link this shared SDL3.framework instead of vcpkg's static SDL3")
if(KRKR_SDL3_FRAMEWORK AND NOT (IOS AND KRKR_HOST_LIBRARY AND USE_SDL3))
    message(FATAL_ERROR "KRKR_SDL3_FRAMEWORK requires an iOS SDL3 build with KRKR_HOST_LIBRARY=ON")
endif()
# 宿主共享的 SDL 生命周期层（Mikage-Next 的 Engine/MikageSDLHost）。iOS 宿主框架
# 把它编译进来，由它驱动 SDL 主回调、音频会话和每局之间的复位。
set(MIKAGE_SDL_HOST_DIR "" CACHE PATH "iOS host: Mikage's shared SDL host sources (Engine/MikageSDLHost)")
if(IOS AND KRKR_HOST_LIBRARY AND NOT EXISTS "${MIKAGE_SDL_HOST_DIR}/MikageSDLHost.h")
    message(FATAL_ERROR "KRKR_HOST_LIBRARY requires MIKAGE_SDL_HOST_DIR to point at Mikage's Engine/MikageSDLHost")
endif()

# 建立项目
if(ONLYCONSOLE)
    project(tjs2 VERSION "0.0.6" LANGUAGES C CXX)
else()
    if(IOS OR MACOS)
        project(krkrsdl3 VERSION "0.0.6" LANGUAGES C CXX OBJC OBJCXX)
    else()
        project(krkrsdl3 VERSION "0.0.6" LANGUAGES C CXX)
    endif()
endif()

set(CMAKE_C_STANDARD 17)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(IOS OR MACOS)
    set(CMAKE_OBJCXX_STANDARD 17)
    set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
endif()
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

if(ANDROID)
    add_definitions(-D_KRKRSDL3_ANDROID=1)
elseif(LINUX)
    add_definitions(-D_KRKRSDL3_LINUX=1)
elseif(EMSCRIPTEN)
    add_definitions(-D_KRKRSDL3_EMSCRIPTEN=1)
elseif(WINDOWS)
    add_definitions(-D_KRKRSDL3_WINDOWS=1)
elseif(OHOS)
    add_definitions(-D_KRKRSDL3_OHOS=1)
elseif(MACOS)
    add_definitions(-D_KRKRSDL3_MACOS=1)
elseif(IOS)
    add_definitions(-D_KRKRSDL3_IOS=1)
else()
    message(FATAL_ERROR "unsupported: ${CMAKE_SYSTEM_NAME}")
endif()
if(USE_RENDER_GL)
    if(ANDROID OR EMSCRIPTEN OR OHOS OR IOS)
        add_definitions(-D_KRKRSDL3_GLES=1)
    elseif(LINUX OR WINDOWS OR MACOS)
        add_definitions(-D_KRKRSDL3_GL=1)
    endif()
endif()

#
# 2. 区分编译
#

if(ONLYCONSOLE)
    include(${CMAKE_CURRENT_SOURCE_DIR}/cmake/cmake_console.txt)
else()
    include(${CMAKE_CURRENT_SOURCE_DIR}/cmake/cmake_main.txt)
endif()
